// Use HIGH accuracy + allow drag correction let savedPin = null; function initMap() { navigator.geolocation.getCurrentPosition(pos => { let lat = pos.coords.latitude; let lng = pos.coords.longitude; // Check if user already corrected pin let corrected = JSON.parse(localStorage.getItem('zondi_corrected_pin')||'null'); if(corrected){ lat = corrected.lat; lng = corrected.lng; } map.setView([lat,lng], 19); // Draggable marker - client can drag to correct house savedPin = L.marker([lat,lng], {draggable:true}).addTo(map) .bindPopup("Drag me to your EXACT gate - then Save") .openPopup(); savedPin.on('dragend', e => { let p = e.target.getLatLng(); localStorage.setItem('zondi_corrected_pin', JSON.stringify({lat:p.lat, lng:p.lng})); localStorage.setItem('zondi_address', p.lat+','+p.lng); // save exact, not street name fetch('/api/sos', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({email: localStorage.getItem('zondi_email'), lat:p.lat, lng:p.lng, corrected:true })}) }); }, null, {enableHighAccuracy:true, maximumAge:0, timeout:10000}); }
